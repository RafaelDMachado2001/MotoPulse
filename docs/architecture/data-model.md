# Modelo de dados e DDL

O bloco é executável sem adaptação (`PRAGMA foreign_keys=ON`). Todas as tabelas são `STRICT`; UUID e timestamps são TEXT. Timestamps externos têm `original_offset` quando conhecido; a sessão pode ter tempo desconhecido. Ausência de medição/valor monetário exige motivo; opcionais estruturais são VIN, apelido, notas, FKs de escopo, períodos de relatório e metadados de detecção.

```sql
PRAGMA foreign_keys=ON;
CREATE TABLE motorcycle(id TEXT NOT NULL PRIMARY KEY,make TEXT NOT NULL,model TEXT NOT NULL,year INTEGER NOT NULL CHECK(year BETWEEN 1900 AND 2200),vin TEXT,nickname TEXT,created_at TEXT NOT NULL,updated_at TEXT NOT NULL) STRICT;
CREATE TABLE odometer_reading(id TEXT NOT NULL PRIMARY KEY,motorcycle_id TEXT NOT NULL REFERENCES motorcycle(id) ON DELETE RESTRICT,read_at TEXT NOT NULL,original_offset TEXT,value_m INTEGER NOT NULL CHECK(value_m>=0),source TEXT NOT NULL CHECK(source IN ('user','maintenance','trip','import')),note TEXT,is_correction INTEGER NOT NULL DEFAULT 0 CHECK(is_correction IN(0,1)),corrects_reading_id TEXT REFERENCES odometer_reading(id) ON DELETE RESTRICT,created_at TEXT NOT NULL,CHECK(is_correction=0 OR(corrects_reading_id IS NOT NULL AND note IS NOT NULL))) STRICT;
CREATE TABLE maintenance_record(id TEXT NOT NULL PRIMARY KEY,motorcycle_id TEXT NOT NULL REFERENCES motorcycle(id) ON DELETE RESTRICT,occurred_at TEXT NOT NULL,original_offset TEXT,type TEXT NOT NULL CHECK(type IN('service','repair','inspection','other')),description TEXT NOT NULL,odometer_reading_id TEXT REFERENCES odometer_reading(id) ON DELETE RESTRICT,cost_cents INTEGER,currency TEXT,cost_missing_reason TEXT,notes TEXT,created_at TEXT NOT NULL,CHECK(cost_cents IS NOT NULL OR cost_missing_reason IS NOT NULL),CHECK((cost_cents IS NULL)=(currency IS NULL)),CHECK(cost_cents IS NULL OR cost_cents>=0)) STRICT;
CREATE TABLE maintenance_item(id TEXT NOT NULL PRIMARY KEY,record_id TEXT NOT NULL REFERENCES maintenance_record(id) ON DELETE RESTRICT,name TEXT NOT NULL,quantity REAL,part_number TEXT,cost_cents INTEGER,currency TEXT,cost_missing_reason TEXT,CHECK(cost_cents IS NOT NULL OR cost_missing_reason IS NOT NULL),CHECK((cost_cents IS NULL)=(currency IS NULL)),CHECK(cost_cents IS NULL OR cost_cents>=0)) STRICT;
CREATE TABLE maintenance_plan(id TEXT NOT NULL PRIMARY KEY,motorcycle_id TEXT NOT NULL REFERENCES motorcycle(id) ON DELETE RESTRICT,name TEXT NOT NULL,interval_m INTEGER,interval_days INTEGER,active INTEGER NOT NULL DEFAULT 1 CHECK(active IN(0,1)),created_at TEXT NOT NULL,CHECK(interval_m IS NOT NULL OR interval_days IS NOT NULL),CHECK(interval_m IS NULL OR interval_m>0),CHECK(interval_days IS NULL OR interval_days>0)) STRICT;
CREATE TABLE plan_item(plan_id TEXT NOT NULL REFERENCES maintenance_plan(id) ON DELETE RESTRICT,name TEXT NOT NULL,part_number TEXT,PRIMARY KEY(plan_id,name)) STRICT;
CREATE TABLE record_plan(record_id TEXT NOT NULL REFERENCES maintenance_record(id) ON DELETE RESTRICT,plan_id TEXT NOT NULL REFERENCES maintenance_plan(id) ON DELETE RESTRICT,PRIMARY KEY(record_id,plan_id)) STRICT;
CREATE TABLE trip(id TEXT NOT NULL PRIMARY KEY,motorcycle_id TEXT NOT NULL REFERENCES motorcycle(id) ON DELETE RESTRICT,started_at TEXT,ended_at TEXT,started_original_offset TEXT,ended_original_offset TEXT,status TEXT NOT NULL CHECK(status IN('draft','recording','paused','interrupted','finished','cancelled')),source TEXT NOT NULL CHECK(source IN('gps','import','demo')),distance_m INTEGER,distance_missing_reason TEXT,wheel_distance_m INTEGER,wheel_distance_missing_reason TEXT,rideology_distance_m INTEGER,rideology_distance_missing_reason TEXT,duration_s INTEGER,duration_missing_reason TEXT,summary_json TEXT,created_at TEXT NOT NULL,updated_at TEXT NOT NULL,CHECK(distance_m IS NOT NULL OR distance_missing_reason IS NOT NULL),CHECK(wheel_distance_m IS NOT NULL OR wheel_distance_missing_reason IS NOT NULL),CHECK(rideology_distance_m IS NOT NULL OR rideology_distance_missing_reason IS NOT NULL)) STRICT;
CREATE TABLE trip_segment(id TEXT NOT NULL PRIMARY KEY,trip_id TEXT NOT NULL REFERENCES trip(id) ON DELETE RESTRICT,sequence INTEGER NOT NULL CHECK(sequence>=0),started_at TEXT NOT NULL,ended_at TEXT,status TEXT NOT NULL CHECK(status IN('recording','paused','finished','interrupted')),UNIQUE(trip_id,sequence)) STRICT;
CREATE TABLE location_point(id TEXT NOT NULL PRIMARY KEY,segment_id TEXT NOT NULL REFERENCES trip_segment(id) ON DELETE RESTRICT,sequence INTEGER NOT NULL CHECK(sequence>=0),recorded_at TEXT NOT NULL,original_offset TEXT NOT NULL,monotonic_ms INTEGER,latitude REAL NOT NULL CHECK(latitude BETWEEN -90 AND 90),longitude REAL NOT NULL CHECK(longitude BETWEEN -180 AND 180),accuracy_m REAL,accuracy_missing_reason TEXT,altitude_m REAL,altitude_missing_reason TEXT,speed_mps REAL,speed_missing_reason TEXT,source TEXT NOT NULL CHECK(source IN('gps_app','rideology_csv','demo')),validity TEXT NOT NULL CHECK(validity IN('valid','invalid','suspect_freeze')),invalid_reason TEXT,CHECK(validity<>'invalid' OR invalid_reason IS NOT NULL),CHECK(validity='valid' OR accuracy_m IS NOT NULL OR source='rideology_csv'),UNIQUE(segment_id,sequence)) STRICT;
CREATE TABLE telemetry_import(id TEXT NOT NULL PRIMARY KEY,motorcycle_id TEXT NOT NULL REFERENCES motorcycle(id) ON DELETE RESTRICT,source_file_name TEXT NOT NULL,source_hash TEXT NOT NULL UNIQUE,raw_uri TEXT NOT NULL,imported_at TEXT NOT NULL,encoding TEXT,delimiter TEXT,decimal_rules_json TEXT,metadata_json TEXT,parser_version TEXT NOT NULL,status TEXT NOT NULL CHECK(status IN('pending','parsed','rejected','failed')),error_summary TEXT) STRICT;
CREATE TABLE telemetry_parse_run(id TEXT NOT NULL PRIMARY KEY,import_id TEXT NOT NULL REFERENCES telemetry_import(id) ON DELETE RESTRICT,parser_version TEXT NOT NULL,status TEXT NOT NULL CHECK(status IN('pending','parsed','rejected','failed')),started_at TEXT NOT NULL,finished_at TEXT,error_summary TEXT,UNIQUE(import_id,parser_version)) STRICT;
CREATE TABLE telemetry_session(id TEXT NOT NULL PRIMARY KEY,motorcycle_id TEXT NOT NULL REFERENCES motorcycle(id) ON DELETE RESTRICT,trip_id TEXT,parse_run_id TEXT,started_at TEXT,started_at_precision_s INTEGER,original_offset TEXT,time_source TEXT NOT NULL CHECK(time_source IN('csv_title','file_name','user','unknown')),started_missing_reason TEXT,ended_at TEXT,source TEXT NOT NULL CHECK(source IN('rideology_csv','ble','demo')),FOREIGN KEY(trip_id,motorcycle_id) REFERENCES trip(id,motorcycle_id) ON DELETE RESTRICT,FOREIGN KEY(parse_run_id) REFERENCES telemetry_parse_run(id) ON DELETE RESTRICT,CHECK(source<>'rideology_csv' OR parse_run_id IS NOT NULL),CHECK(started_at IS NOT NULL OR started_missing_reason IS NOT NULL)) STRICT;
CREATE TABLE telemetry_sample(id TEXT NOT NULL PRIMARY KEY,session_id TEXT NOT NULL REFERENCES telemetry_session(id) ON DELETE RESTRICT,row_number INTEGER,elapsed_ms INTEGER NOT NULL CHECK(elapsed_ms>=0),recorded_at TEXT,original_offset TEXT,recorded_at_missing_reason TEXT,rpm REAL,speed_mps REAL,gear TEXT CHECK(gear IN('N','1','2','3','4','5','6')),fuel_km_per_l REAL,engine_temp_c REAL,latitude REAL,longitude REAL,values_json TEXT,missing_fields_json TEXT,CHECK(row_number IS NOT NULL OR missing_fields_json IS NOT NULL),UNIQUE(session_id,elapsed_ms)) STRICT;
CREATE TABLE sample_issue(id TEXT NOT NULL PRIMARY KEY,import_id TEXT NOT NULL REFERENCES telemetry_import(id) ON DELETE RESTRICT,parse_run_id TEXT REFERENCES telemetry_parse_run(id) ON DELETE RESTRICT,session_id TEXT REFERENCES telemetry_session(id) ON DELETE RESTRICT,sample_id TEXT REFERENCES telemetry_sample(id) ON DELETE RESTRICT,row_number INTEGER,field TEXT,code TEXT NOT NULL,detail TEXT NOT NULL,CHECK(row_number IS NOT NULL OR field='file')) STRICT;
CREATE TABLE metric_result(id TEXT NOT NULL PRIMARY KEY,trip_id TEXT REFERENCES trip(id) ON DELETE RESTRICT,session_id TEXT REFERENCES telemetry_session(id) ON DELETE RESTRICT,metric_type TEXT NOT NULL,value REAL,missing_reason TEXT,unit TEXT NOT NULL,method_version TEXT NOT NULL,calculation_key TEXT NOT NULL,coverage_json TEXT NOT NULL,evidence_json TEXT NOT NULL,calculated_at TEXT NOT NULL,CHECK((trip_id IS NULL)<>(session_id IS NULL)),CHECK(value IS NOT NULL OR missing_reason IS NOT NULL)) STRICT;
CREATE TABLE baseline_profile(id TEXT NOT NULL PRIMARY KEY,motorcycle_id TEXT NOT NULL REFERENCES motorcycle(id) ON DELETE RESTRICT,metric_type TEXT NOT NULL,cohort_definition_json TEXT NOT NULL,sample_count INTEGER NOT NULL CHECK(sample_count>=0),p50 REAL,p90 REAL,mean REAL,stddev REAL,missing_reason TEXT,version TEXT NOT NULL,calculated_at TEXT NOT NULL,CHECK(p50 IS NOT NULL OR p90 IS NOT NULL OR mean IS NOT NULL OR stddev IS NOT NULL OR missing_reason IS NOT NULL),UNIQUE(motorcycle_id,metric_type,version)) STRICT;
CREATE TABLE finding(id TEXT NOT NULL PRIMARY KEY,motorcycle_id TEXT NOT NULL REFERENCES motorcycle(id) ON DELETE RESTRICT,trip_id TEXT REFERENCES trip(id) ON DELETE RESTRICT,session_id TEXT REFERENCES telemetry_session(id) ON DELETE RESTRICT,baseline_profile_id TEXT REFERENCES baseline_profile(id) ON DELETE RESTRICT,metric_result_id TEXT REFERENCES metric_result(id) ON DELETE RESTRICT,metric_type TEXT NOT NULL,kind TEXT NOT NULL CHECK(kind IN('measurement','deviation','hypothesis','recommendation')),observed_value REAL,expected_value REAL,deviation REAL,confidence REAL,missing_reason TEXT,evidence_json TEXT NOT NULL,classification TEXT NOT NULL,algorithm_version TEXT NOT NULL,created_at TEXT NOT NULL,CHECK(observed_value IS NOT NULL OR missing_reason IS NOT NULL),CHECK(kind<>'deviation' OR baseline_profile_id IS NOT NULL)) STRICT;
CREATE TABLE report(id TEXT NOT NULL PRIMARY KEY,motorcycle_id TEXT NOT NULL REFERENCES motorcycle(id) ON DELETE RESTRICT,scope TEXT NOT NULL CHECK(scope IN('trip','period','maintenance')),trip_id TEXT REFERENCES trip(id) ON DELETE RESTRICT,period_start TEXT,period_end TEXT,structured_input_json TEXT NOT NULL,input_hash TEXT NOT NULL,generation_version TEXT NOT NULL,narrative TEXT NOT NULL,uncertainty_json TEXT NOT NULL,generated_at TEXT NOT NULL,CHECK(scope<>'trip' OR trip_id IS NOT NULL),CHECK(scope<>'period' OR(period_start IS NOT NULL AND period_end IS NOT NULL))) STRICT;
CREATE UNIQUE INDEX ux_motorcycle_vin ON motorcycle(make,model,year,vin) WHERE vin IS NOT NULL;
CREATE UNIQUE INDEX ux_trip_active ON trip(motorcycle_id) WHERE status IN('recording','paused','interrupted');
CREATE UNIQUE INDEX ux_metric_trip ON metric_result(trip_id,metric_type,method_version,calculation_key) WHERE trip_id IS NOT NULL;
CREATE UNIQUE INDEX ux_metric_session ON metric_result(session_id,metric_type,method_version,calculation_key) WHERE session_id IS NOT NULL;
CREATE UNIQUE INDEX ux_session_trip_moto ON telemetry_session(id,motorcycle_id);
CREATE UNIQUE INDEX ux_trip_moto ON trip(id,motorcycle_id);
CREATE INDEX ix_sample_session_elapsed ON telemetry_sample(session_id,elapsed_ms);
CREATE INDEX ix_issue_import_row ON sample_issue(import_id,row_number);
CREATE TRIGGER odometer_same_moto BEFORE INSERT ON odometer_reading WHEN NEW.is_correction=1 AND (SELECT motorcycle_id FROM odometer_reading WHERE id=NEW.corrects_reading_id)<>NEW.motorcycle_id BEGIN SELECT RAISE(ABORT,'correction motorcycle mismatch'); END;
CREATE TRIGGER raw_location_immutable BEFORE UPDATE ON location_point BEGIN SELECT RAISE(ABORT,'raw location immutable'); END;
CREATE TRIGGER raw_sample_immutable BEFORE UPDATE ON telemetry_sample BEGIN SELECT RAISE(ABORT,'raw sample immutable'); END;
```

O CSV real usa CP932, metadados de preâmbulo e 14 colunas; o importador escolhe encoding por arquivo, regra decimal por coluna e registra `metadata_json`, sem supor este formato para todo arquivo. `elapsed_ms`, km/L, N, lat/lon e constantes ficam como lidos; L/100km é derivado. 0.0/99.9 são [A VERIFICAR]. GPS CSV sem precisão é válido por fonte; coordenada repetida em movimento vira `suspect_freeze`, não invalida automaticamente. Hash global deduplica bytes; parse runs permitem nova versão. Exclusão de bruto é decisão aberta; por ora RESTRICT. Correção de odômetro é garantida por trigger e teste.
